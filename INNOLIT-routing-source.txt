import { useEffect, useState } from 'react'
import {
  Lightbulb, Plus, PanelRightClose,
  PanelRightOpen, FolderOpen, Trash2
} from 'lucide-react'
import { ConceptStudio } from './ConceptStudio'
import { CreativeChat } from './CreativeChat'
import type { ChatIdea, ConceptStudioState } from './concepts'
import type { ModelSettings } from './modelGeneration'
import type { ImageSettings } from './imageGeneration'
import { deleteProjectVisuals } from './visualStore'
import { createProject, getProjectStudio, loadProjects, STORAGE_KEY } from './projects'
import type { Message } from './projects'
import './App.css'

function App() {
  const [projects, setProjects] = useState(loadProjects)
  const [activeId, setActiveId] = useState(
    () => projects[0].id
  )
  const [input, setInput] = useState('')
  const [showResults, setShowResults] = useState(true)
  const [view, setView] = useState<'workspace' | 'studio'>('workspace')
  const [saveFailed, setSaveFailed] = useState(false)
  const [modelSettings, setModelSettings] = useState<ModelSettings>({ baseUrl: '', model: '', apiKey: '' })
  const [imageSettings, setImageSettings] = useState<ImageSettings>({ baseUrl: '', model: '', apiKey: '', protocol: 'siliconflow', size: '1024x1024' })
  const [transfer, setTransfer] = useState<{ projectId: string; idea: ChatIdea } | null>(null)

  const activeProject =
    projects.find(project => project.id === activeId)
    ?? projects[0]

  useEffect(() => {
    try {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(projects))
      // Report the outcome of writing to browser storage.
      // oxlint-disable-next-line react/set-state-in-effect
      setSaveFailed(false)
    } catch {
      // oxlint-disable-next-line react/set-state-in-effect
      setSaveFailed(true)
    }
  }, [projects])

  function updateStudio(update: (current: ConceptStudioState) => ConceptStudioState) {
    const updatedAt = Date.now()
    setProjects(current => current.map(project => project.id === activeProject.id
      ? { ...project, conceptStudio: update(getProjectStudio(project)), updatedAt }
      : project))
  }

  function editModelSettings(change: Partial<ModelSettings>) {
    setModelSettings(current => {
      const next = { ...current, ...change }
      if (next.baseUrl.trim() !== current.baseUrl.trim()) next.apiKey = ''
      return next
    })
  }

  function addProject() {
    const project = createProject()
    setProjects(current => [project, ...current])
    setActiveId(project.id)
    setInput('')
  }

  function editImageSettings(change: Partial<ImageSettings>) {
    setImageSettings(current => {
      const next = { ...current, ...change }
      if (next.baseUrl.trim() !== current.baseUrl.trim()) next.apiKey = ''
      return next
    })
  }

  function deleteProject(id: string) {
    // Removing a project must still work when browser image storage is unavailable.
    void deleteProjectVisuals(id).catch(() => {})
    const replacement = createProject()
    setProjects(current => {
      const remaining = current.filter(project => project.id !== id)
      return remaining.length ? remaining : [replacement]
    })

    if (activeId === id) {
      setActiveId(projects.find(project => project.id !== id)?.id ?? replacement.id)
      setInput('')
    }
  }

  function appendChat(projectId: string, user: Message, assistant: Message) {
    const updatedAt = Date.now()
    setProjects(current => current.map(project => {
      if (project.id !== projectId) return project

      return {
        ...project,
        name: project.messages.length === 0
          ? user.content.slice(0, 24)
          : project.name,
        messages: [...project.messages, user, assistant],
        updatedAt
      }
    }))
  }

  const latestIdea = [...activeProject.messages]
    .reverse()
    .find(message => message.role === 'user')

  return (
    <div className="app-shell">
      <aside className="sidebar">
        <div className="brand">
          <span className="brand-mark">
            <Lightbulb size={19} />
          </span>
          <span>INNO LIT</span>
        </div>

        <button
          className="new-project"
          onClick={addProject}
          aria-label="新建设计项目"
        >
          <Plus size={17} />
          新建设计项目
        </button>

        <div className="section-label">我的项目</div>

        <div className="project-list">
          {projects.map(project => (
            <div
              className={
                'project-item ' +
                (project.id === activeId ? 'active' : '')
              }
              key={project.id}
            >
              <button
                className="project-select"
                aria-label={project.name}
                onClick={() => {
                  setActiveId(project.id)
                  setInput('')
                }}
              >
                <FolderOpen size={16} />
                <span>{project.name}</span>
              </button>

              <button
                className="delete-project"
                title="删除项目"
                aria-label={`删除项目 ${project.name}`}
                onClick={() => {
                  if (window.confirm(
                    `确定删除“${project.name}”吗？`
                  )) {
                    deleteProject(project.id)
                  }
                }}
              >
                <Trash2 size={15} />
              </button>
            </div>
          ))}
        </div>

        <div className="sidebar-footer">
          V0.2 · Local Workspace
        </div>
      </aside>

      <main className="workspace">
        <header className="topbar">
          <div className="breadcrumb">
            工作台 <span>/</span> {activeProject.name}
          </div>

          {view === 'workspace' && (
            <button
              className="icon-button"
              title="切换成果面板"
              aria-label="切换成果面板"
              aria-expanded={showResults}
              onClick={() => setShowResults(value => !value)}
            >
              {showResults
                ? <PanelRightClose size={19} />
                : <PanelRightOpen size={19} />}
            </button>
          )}
        </header>

        <nav className="workspace-nav" aria-label="项目工作区">
          <button type="button" aria-pressed={view === 'workspace'} onClick={() => setView('workspace')}>创意对话</button>
          <button type="button" aria-pressed={view === 'studio'} onClick={() => setView('studio')}>Concept Studio</button>
        </nav>
        {saveFailed && <p className="workspace-save-error" role="alert">当前浏览器无法保存项目，刷新可能丢失本次更改。</p>}

        {view === 'studio' ? (
          <ConceptStudio
            key={activeProject.id}
            projectId={activeProject.id}
            projectName={activeProject.name}
            state={getProjectStudio(activeProject)}
            onChange={updateStudio}
            settings={modelSettings}
            onSettingsChange={editModelSettings}
            imageSettings={imageSettings}
            onImageSettingsChange={editImageSettings}
            idea={transfer?.projectId === activeProject.id ? transfer.idea : undefined}
            onDismissIdea={() => setTransfer(null)}
          />
        ) : (
          <CreativeChat
            key={activeProject.id}
            messages={activeProject.messages}
            input={input}
            onInputChange={setInput}
            onAppend={(user, assistant) => appendChat(activeProject.id, user, assistant)}
            onTransfer={idea => {
              setTransfer({ projectId: activeProject.id, idea })
              setView('studio')
            }}
            settings={modelSettings}
            onSettingsChange={editModelSettings}
          />
        )}
      </main>

      {view === 'workspace' && showResults && (
        <aside className="results-panel">
          <div className="results-header">
            <h2>设计成果</h2>
            <span>Workspace</span>
          </div>

          <div className="result-section">
            <div className="section-label">当前概念</div>
            {latestIdea ? (
              <div className="concept-card">
                <div className="concept-icon">
                  <Lightbulb size={18} />
                </div>
                <h3>{activeProject.name}</h3>
                <p>{latestIdea.content}</p>
                <span className="concept-status">
                  初步构思
                </span>
              </div>
            ) : (
              <div className="empty-result">
                输入设计想法后，概念摘要将在这里出现。
              </div>
            )}
          </div>

          <div className="result-section">
            <div className="section-label">设计流程</div>
            <div className="process-list">
              <div>01 · 问题与场景</div>
              <div>02 · 创意方向</div>
              <div>03 · 产品概念</div>
              <div>04 · 可行性验证与原创性检索</div>
              <div>05 · 设计报告</div>
            </div>
          </div>

          <div className="results-footer">
            项目内容保存在当前浏览器
          </div>
        </aside>
      )}
    </div>
  )
}

export default App

import { useEffect, useRef, useState } from 'react'
import type { FormEvent } from 'react'
import { ArrowLeft, ArrowRight, Layers } from 'lucide-react'
import {
  addChatConcept, appendConceptRound, conceptCounts, conceptSourceLabel, draftFromConcept, normalizeDraft, sentenceLabels,
  startingPointLimit,
  modelSourceLabel,
} from './concepts'
import type {
  ChatIdea, Concept, ConceptBatch, ConceptCount, ConceptDraft, ConceptStudioState, Score, ScoreDimension,
} from './concepts'
import { RefinementWorkspace } from './RefinementWorkspace'
import { canUseCustomModel, generateWithModel, hasModelSettings } from './modelGeneration'
import type { ModelSettings } from './modelGeneration'
import type { ImageSettings } from './imageGeneration'
import { ModelControls } from './ModelControls'
import { ChatConceptImport } from './ChatConceptImport'
import { ConceptComparison } from './ConceptComparison'
import './ConceptStudio.css'

type StudioProps = {
  projectId: string
  projectName: string
  state: ConceptStudioState
  onChange: (update: (current: ConceptStudioState) => ConceptStudioState) => void
  settings: ModelSettings
  onSettingsChange: (change: Partial<ModelSettings>) => void
  imageSettings: ImageSettings
  onImageSettingsChange: (change: Partial<ImageSettings>) => void
  idea?: ChatIdea
  onDismissIdea: () => void
}

const sentences = Object.keys(sentenceLabels) as (keyof typeof sentenceLabels)[]

function ConceptFramework({ concept }: { concept: Concept }) {
  return (
    <ol className="concept-framework">
      {sentences.map(key => (
        <li key={key}>
          <strong>{sentenceLabels[key]}</strong>
          <span>{concept.sentences[key]}</span>
        </li>
      ))}
    </ol>
  )
}

function focusHeading(heading: HTMLHeadingElement | null) {
  heading?.scrollIntoView({ block: 'start' })
  heading?.focus({ preventScroll: true })
}

export function ConceptStudio({ projectId, projectName, state, onChange, settings, onSettingsChange, imageSettings, onImageSettingsChange, idea, onDismissIdea }: StudioProps) {
  const [refining, setRefining] = useState(false)
  const [customModel, setCustomModel] = useState(false)
  const [generating, setGenerating] = useState(false)
  const [notice, setNotice] = useState('')
  const request = useRef<AbortController | null>(null)
  const briefHeading = useRef<HTMLHeadingElement>(null)
  const conceptHeading = useRef<HTMLHeadingElement>(null)
  const comparisonHeading = useRef<HTMLHeadingElement>(null)
  const pendingFocus = useRef<'brief' | 'concepts' | 'comparison' | null>(null)
  const { draft, rounds, activeRound } = state
  const batch = rounds[activeRound]
  const latest = rounds.at(-1)
  const selected = batch?.concepts.find(concept => concept.id === batch.selectedId)
  const normalized = normalizeDraft(draft)
  const ready = normalized.problem && normalized.audience && normalized.scenario
  const configured = hasModelSettings(settings)
  const unchanged = latest && !latest.origin && !latest.generation && !customModel && (
    latest.brief.problem === normalized.problem &&
    latest.brief.audience === normalized.audience &&
    latest.brief.scenario === normalized.scenario &&
    latest.brief.count === normalized.count &&
    latest.brief.startingPoint?.text === normalized.startingPoint?.text &&
    latest.brief.startingPoint?.source?.conceptId === normalized.startingPoint?.source?.conceptId
  )

  useEffect(() => {
    if (!pendingFocus.current) return
    const heading = pendingFocus.current === 'brief' ? briefHeading
      : pendingFocus.current === 'concepts' ? conceptHeading : comparisonHeading
    pendingFocus.current = null
    focusHeading(heading.current)
  }, [rounds.length, batch?.concepts.length, activeRound, draft, refining, idea])

  useEffect(() => () => request.current?.abort(), [])

  function editDraft(fields: Partial<ConceptDraft>) {
    setNotice('')
    onChange(current => ({ ...current, draft: { ...current.draft, ...fields } }))
  }

  async function generate(event: FormEvent<HTMLFormElement>) {
    event.preventDefault()
    if (!ready || unchanged || request.current) return
    setNotice('')
    if (!customModel) {
      pendingFocus.current = 'concepts'
      onChange(appendConceptRound)
      return
    }
    if (!canUseCustomModel(window.location.hostname) || !configured) return

    const controller = new AbortController()
    request.current = controller
    setGenerating(true)
    try {
      const signal = AbortSignal.any([controller.signal, AbortSignal.timeout(150000)])
      const result = await generateWithModel(draft, settings, signal)
      
      if (controller.signal.aborted) return
      pendingFocus.current = 'concepts'
      onChange(current => appendConceptRound(current, result))
    } catch (error) {
      if (!controller.signal.aborted) {
        setNotice(error instanceof DOMException && error.name === 'TimeoutError'
          ? '等待模型服务超时。可减少概念数量后重试。'
          : error instanceof Error && !(error instanceof TypeError)
            ? error.message
            : '无法连接模型服务。请检查配置，或选择本地规则。')
      }
    } finally {
      if (request.current === controller) {
        request.current = null
        if (!controller.signal.aborted) setGenerating(false)
      }
    }
  }

  function cancelGeneration() {
    request.current?.abort()
    request.current = null
    setGenerating(false)
    setNotice('已取消本轮生成。')
  }

  function updateRound(update: (round: ConceptBatch) => ConceptBatch) {
    onChange(current => ({
      ...current,
      rounds: current.rounds.map((round, index) => index === activeRound ? update(round) : round),
    }))
  }

  function editConcept(id: Concept['id'], update: (concept: Concept) => Concept) {
    updateRound(round => ({
      ...round,
      concepts: round.concepts.map(concept => concept.id === id ? update(concept) : concept),
    }))
  }

  function setScore(id: Concept['id'], dimension: ScoreDimension, score: Score | null) {
    editConcept(id, concept => ({ ...concept, scores: { ...concept.scores, [dimension]: score } }))
  }

  function selectConcept(id: Concept['id']) {
    updateRound(round => ({ ...round, selectedId: id }))
  }

  if (refining && batch && selected) {
    return (
      <section className="studio-scroll" key="refinement" aria-labelledby="refinement-heading">
        <div className="studio-content">
          <button type="button" className="studio-button" onClick={() => {
            pendingFocus.current = 'comparison'
            setRefining(false)
          }}>
            <ArrowLeft size={16} aria-hidden="true" />
            返回概念比较
          </button>
          <header className="studio-intro">
            <p className="studio-kicker">第 {activeRound + 1} 轮 · 已选方案 · {selected.development?.frozen?.design.format ?? selected.format}</p>
            <h1 id="refinement-heading" ref={focusHeading} tabIndex={-1}>方案深化 · {selected.development?.frozen?.design.title ?? selected.title}</h1>
            <p>AI 协助深化、验证与检索，经你修正后完善概念并冻结，再制作一致的三张展示图。</p>
            <span className="studio-badge">初始概念：{conceptSourceLabel(batch)}</span>
            {selected.development?.frozen && <span className="studio-badge">完善模型：{modelSourceLabel(selected.development.frozen.source)}</span>}
          </header>
          <RefinementWorkspace
            key={`${activeRound}:${selected.id}`}
            projectId={projectId}
            projectName={projectName}
            round={batch}
            roundNumber={activeRound + 1}
            concept={selected}
            onChange={update => editConcept(selected.id, update)}
            imageSettings={imageSettings}
            onImageSettingsChange={onImageSettingsChange}
            settings={settings}
            onSettingsChange={onSettingsChange}
          >
            <details>
              <summary>查看完整九句话</summary>
              <ConceptFramework concept={selected} />
            </details>
          </RefinementWorkspace>
        </div>
      </section>
    )
  }

  return (
    <section className="studio-scroll" key="studio" aria-labelledby="studio-heading">
      <div className="studio-content">
        <header className="studio-intro">
          <p className="studio-kicker"><Layers size={15} aria-hidden="true" /> V0.2 · 创意方案</p>
          <h1 id="studio-heading">Concept Studio</h1>
          <p>从同一设计问题探索 3–6 个方向，再比较、评分并选定一个进行深化。</p>
        </header>

        {idea && (
          <ChatConceptImport
            key={idea.messageId}
            idea={idea}
            state={state}
            settings={settings}
            onDismiss={onDismissIdea}
            onSave={(destination, brief, candidate, useAsDraft) => {
              pendingFocus.current = useAsDraft ? 'brief' : 'concepts'
              onChange(current => addChatConcept(current, destination, brief, candidate, useAsDraft))
              if (useAsDraft) {
                setNotice('已加入概念并填入简报草稿。核对设计起点、目标用户和使用场景后即可生成。')
              }
              onDismissIdea()
            }}
          />
        )}

        <form className="studio-surface studio-brief" onSubmit={generate}>
          <h2 ref={briefHeading} tabIndex={-1}>生成设计方向</h2>
          <label className="studio-field studio-problem">
            你想设计什么？
            <textarea
              rows={2}
              value={draft.problem}
              onChange={event => editDraft({ problem: event.target.value })}
              placeholder="例如，如何帮助学生在课间快速放松？"
              disabled={generating}
              required
            />
          </label>
          <label className="studio-field">
            目标用户
            <input
              value={draft.audience}
              onChange={event => editDraft({ audience: event.target.value })}
              placeholder="例如，备考的高中生"
              disabled={generating}
              required
            />
          </label>
          <div className="studio-starting-point">
            <label className="studio-field">
              设计起点（可选）
              <textarea
                rows={3}
                maxLength={startingPointLimit}
                value={draft.startingPoint?.text ?? ''}
                placeholder="填写希望延伸的创意、机制或原型方向"
                disabled={generating}
                aria-describedby={draft.startingPoint?.source ? 'starting-point-source' : undefined}
                onChange={event => {
                  const text = event.target.value
                  editDraft({ startingPoint: text ? { text, source: draft.startingPoint?.source } : undefined })
                }}
              />
            </label>
            {draft.startingPoint?.source && (
              <p className="studio-hint" id="starting-point-source">整理自「{draft.startingPoint.source.title}」</p>
            )}
          </div>
          <label className="studio-field">
            使用场景
            <input
              value={draft.scenario}
              onChange={event => editDraft({ scenario: event.target.value })}
              placeholder="例如，两节课之间的教室桌面"
              disabled={generating}
              required
            />
          </label>
          <div className="studio-model">
            <ModelControls
              settings={settings}
              onSettingsChange={change => {
                onSettingsChange(change)
                setNotice('')
              }}
              customModel={customModel}
              onModeChange={custom => {
                setCustomModel(custom)
                setNotice('')
              }}
              disabled={generating}
              requestDescription="生成时会将当前设计简报发送给所选服务。"
            />
          </div>
          <div className="studio-actions">
            <label className="studio-field">
              生成数量
              <select value={draft.count} disabled={generating} onChange={event => editDraft({ count: Number(event.target.value) as ConceptCount })}>
                {conceptCounts.map(count => <option key={count} value={count}>{count} 个</option>)}
              </select>
            </label>
            <button type="submit" className="studio-button primary" disabled={!ready || Boolean(unchanged) || generating || (customModel && !configured)}>
              {generating ? '正在生成…' : normalized.startingPoint ? '基于简报生成概念' : latest ? '生成新一轮' : '生成概念'}
              <ArrowRight size={16} aria-hidden="true" />
            </button>
            {generating && <button type="button" className="studio-button" onClick={cancelGeneration}>取消生成</button>}
          </div>
          <p className="studio-hint" role="status">
            {generating
              ? `正在使用 ${settings.model.trim()} 生成概念…`
              : notice || (latest
              ? unchanged
                ? `最新的第 ${rounds.length} 轮有 ${latest.concepts.length} 个概念。可修改简报、数量或选择模型生成新一轮。`
                : '生成新一轮会保留已有轮次及其评分和深化笔记。'
              : '填写三个字段后即可生成概念。')}
          </p>
        </form>

        {batch && (
          <>
            <section className="studio-section" aria-labelledby="concept-heading">
              <h2 id="concept-heading" tabIndex={-1} ref={conceptHeading}>概念方向 <span>{batch.concepts.length} 个</span></h2>
              <p className="studio-hint">{conceptSourceLabel(batch)}</p>
              <div className="studio-round-bar">
                <label className="studio-field">
                  概念轮次
                  <select value={activeRound} onChange={event => {
                    const index = Number(event.target.value)
                    onChange(current => ({ ...current, activeRound: index }))
                  }}>
                    {rounds.map((round, index) => (
                      <option key={index} value={index}>
                        第 {index + 1} 轮 · {round.concepts.length} 个概念{index === rounds.length - 1 ? '（最新）' : ''}
                      </option>
                    ))}
                  </select>
                </label>
                <div className="studio-actions">
                  <button type="button" className="studio-button" onClick={() => focusHeading(comparisonHeading.current)}>
                    查看比较 <ArrowRight size={16} aria-hidden="true" />
                  </button>
                  {selected && (
                    <button type="button" className="studio-button primary" disabled={generating} onClick={() => setRefining(true)}>继续深化</button>
                  )}
                </div>
              </div>
              <details className="studio-round-brief" key={`brief-${activeRound}`}>
                <summary>查看本轮生成简报</summary>
                <dl>
                  <dt>设计问题</dt><dd>{batch.brief.problem}</dd>
                  <dt>目标用户</dt><dd>{batch.brief.audience}</dd>
                  <dt>使用场景</dt><dd>{batch.brief.scenario}</dd>
                  {batch.brief.startingPoint && (
                    <>
                      <dt>设计起点</dt><dd>{batch.brief.startingPoint.text}</dd>
                      {batch.brief.startingPoint.source && <><dt>起点概念</dt><dd>{batch.brief.startingPoint.source.title}</dd></>}
                    </>
                  )}
                </dl>
              </details>
              <p className="studio-hint">展开卡片查看每个方向的完整九句话。</p>
              <div className="studio-cards" key={`cards-${activeRound}`}>
                {batch.concepts.map((concept, index) => (
                  <article className="studio-surface studio-card" key={concept.id} data-selected={selected?.id === concept.id}>
                    <p className="studio-kicker">{String(index + 1).padStart(2, '0')} · {concept.format}</p>
                    <h3>{concept.title}</h3>
                    {concept.chatOrigin && <span className="studio-badge">来自创意对话</span>}
                    <p>{concept.sentences.idea}</p>
                    <details>
                      <summary>查看九句话</summary>
                      <ConceptFramework concept={concept} />
                    </details>
                    {concept.chatOrigin && (
                      <button type="button" className="studio-button studio-use-starting-point" disabled={generating} onClick={() => {
                        pendingFocus.current = 'brief'
                        onChange(current => ({ ...current, draft: draftFromConcept(current.draft, batch.brief, concept) }))
                        setNotice('已用这张对话概念卡填入简报草稿。核对后可生成新的变体。')
                      }}>
                        填入简报草稿
                      </button>
                    )}
                  </article>
                ))}
              </div>
            </section>

            <section className="studio-section" aria-labelledby="comparison-heading">
              <h2 id="comparison-heading" tabIndex={-1} ref={comparisonHeading}>概念比较</h2>
              <ConceptComparison

import type { ModelSettings } from './modelGeneration'
import type { ImageSettings } from './imageGeneration'

export const aiTasks = {
  concept: '创意生成',
  refinement: '方案深化',
  translation: '视觉语言转译',
  image: '图像生成',
} as const

export type AITask = keyof typeof aiTasks
export type TextTask = Exclude<AITask, 'image'>

export type AIModelRouting = {
  concept: ModelSettings | null
  refinement: ModelSettings | null
  translation: ModelSettings | null
  image: ImageSettings | null
}

export type ResolvedAIModel = {
  task: AITask
  source: 'dedicated' | 'fallback'
  settings: ModelSettings | ImageSettings
}

export function createEmptyRouting(): AIModelRouting {
  return {
    concept: null,
    refinement: null,
    translation: null,
    image: null,
  }
}

export function resolveTextModel(
  task: TextTask,
  routing: AIModelRouting,
  fallback: ModelSettings,
): ModelSettings {
  return routing[task] ?? fallback
}

export function resolveImageModel(
  routing: AIModelRouting,
  fallback: ImageSettings,
): ImageSettings {
  return routing.image ?? fallback
}

export function resolveAIModel(
  task: AITask,
  routing: AIModelRouting,
  fallbackText: ModelSettings,
  fallbackImage: ImageSettings,
): ResolvedAIModel {
  if (task === 'image') {
    return {
      task,
      source: routing.image ? 'dedicated' : 'fallback',
      settings: resolveImageModel(routing, fallbackImage),
    }
  }

  return {
    task,
    source: routing[task] ? 'dedicated' : 'fallback',
    settings: resolveTextModel(task, routing, fallbackText),
  }
}

export function updateTaskModel<T extends AITask>(
  routing: AIModelRouting,
  task: T,
  settings: AIModelRouting[T],
): AIModelRouting {
  return {
    ...routing,
    [task]: settings,
  }
}

export function clearTaskModel(
  routing: AIModelRouting,
  task: AITask,
): AIModelRouting {
  return {
    ...routing,
    [task]: null,
  }
}

export function taskModelLabel(
  task: AITask,
  routing: AIModelRouting,
  fallbackText: ModelSettings,
  fallbackImage: ImageSettings,
): string {
  const resolved = resolveAIModel(
    task,
    routing,
    fallbackText,
    fallbackImage,
  )

  const model = resolved.settings.model.trim() || '未配置'
  const suffix =
    resolved.source === 'fallback' ? '（默认配置）' : ''

  return `${model}${suffix}`
}
