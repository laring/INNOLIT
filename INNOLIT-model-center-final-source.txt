import { useEffect, useState } from 'react'
import {
  Lightbulb, Plus, PanelRightClose,
  PanelRightOpen, FolderOpen, Trash2
} from 'lucide-react'
import { ConceptStudio } from './ConceptStudio'
import { CreativeChat } from './CreativeChat'
import type { ChatIdea, ConceptStudioState } from './concepts'
import type { ModelSettings } from './modelGeneration'
import type { ImageSettings } from './imageGeneration'
import { createEmptyRouting, resolveTextModel } from './aiModelRouter'
import type { AIModelRouting, AITask } from './aiModelRouter'
import { deleteProjectVisuals } from './visualStore'
import { createProject, getProjectStudio, loadProjects, STORAGE_KEY } from './projects'
import type { Message } from './projects'
import './App.css'

function App() {
  const [projects, setProjects] = useState(loadProjects)
  const [activeId, setActiveId] = useState(
    () => projects[0].id
  )
  const [input, setInput] = useState('')
  const [showResults, setShowResults] = useState(true)
  const [view, setView] = useState<'workspace' | 'studio'>('workspace')
  const [saveFailed, setSaveFailed] = useState(false)
  const [modelSettings, setModelSettings] = useState<ModelSettings>({ baseUrl: '', model: '', apiKey: '' })
  const [imageSettings, setImageSettings] = useState<ImageSettings>({ baseUrl: '', model: '', apiKey: '', protocol: 'siliconflow', size: '1024x1024' })
  const [routing, setRouting] = useState<AIModelRouting>(createEmptyRouting)
  function editTaskSettings(task: AITask, change: Partial<ModelSettings> | Partial<ImageSettings>) {
    setRouting(current => {
      const previous = current[task] ?? (task === 'image' ? imageSettings : modelSettings)
      const next = { ...previous, ...change }
      if (next.baseUrl.trim() !== previous.baseUrl.trim()) next.apiKey = ''
      return { ...current, [task]: next } as AIModelRouting
    })
  }
  function resetTaskSettings(task: AITask) {
    setRouting(current => ({ ...current, [task]: null }))
  }
  const [transfer, setTransfer] = useState<{ projectId: string; idea: ChatIdea } | null>(null)

  const activeProject =
    projects.find(project => project.id === activeId)
    ?? projects[0]

  useEffect(() => {
    try {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(projects))
      // Report the outcome of writing to browser storage.
      // oxlint-disable-next-line react/set-state-in-effect
      setSaveFailed(false)
    } catch {
      // oxlint-disable-next-line react/set-state-in-effect
      setSaveFailed(true)
    }
  }, [projects])

  function updateStudio(update: (current: ConceptStudioState) => ConceptStudioState) {
    const updatedAt = Date.now()
    setProjects(current => current.map(project => project.id === activeProject.id
      ? { ...project, conceptStudio: update(getProjectStudio(project)), updatedAt }
      : project))
  }

  function editModelSettings(change: Partial<ModelSettings>) {
    setModelSettings(current => {
      const next = { ...current, ...change }
      if (next.baseUrl.trim() !== current.baseUrl.trim()) next.apiKey = ''
      return next
    })
  }

  function addProject() {
    const project = createProject()
    setProjects(current => [project, ...current])
    setActiveId(project.id)
    setInput('')
  }

  function editImageSettings(change: Partial<ImageSettings>) {
    setImageSettings(current => {
      const next = { ...current, ...change }
      if (next.baseUrl.trim() !== current.baseUrl.trim()) next.apiKey = ''
      return next
    })
  }

  function deleteProject(id: string) {
    // Removing a project must still work when browser image storage is unavailable.
    void deleteProjectVisuals(id).catch(() => {})
    const replacement = createProject()
    setProjects(current => {
      const remaining = current.filter(project => project.id !== id)
      return remaining.length ? remaining : [replacement]
    })

    if (activeId === id) {
      setActiveId(projects.find(project => project.id !== id)?.id ?? replacement.id)
      setInput('')
    }
  }

  function appendChat(projectId: string, user: Message, assistant: Message) {
    const updatedAt = Date.now()
    setProjects(current => current.map(project => {
      if (project.id !== projectId) return project

      return {
        ...project,
        name: project.messages.length === 0
          ? user.content.slice(0, 24)
          : project.name,
        messages: [...project.messages, user, assistant],
        updatedAt
      }
    }))
  }

  const latestIdea = [...activeProject.messages]
    .reverse()
    .find(message => message.role === 'user')

  return (
    <div className="app-shell">
      <aside className="sidebar">
        <div className="brand">
          <span className="brand-mark">
            <Lightbulb size={19} />
          </span>
          <span>INNO LIT</span>
        </div>

        <button
          className="new-project"
          onClick={addProject}
          aria-label="新建设计项目"
        >
          <Plus size={17} />
          新建设计项目
        </button>

        <div className="section-label">我的项目</div>

        <div className="project-list">
          {projects.map(project => (
            <div
              className={
                'project-item ' +
                (project.id === activeId ? 'active' : '')
              }
              key={project.id}
            >
              <button
                className="project-select"
                aria-label={project.name}
                onClick={() => {
                  setActiveId(project.id)
                  setInput('')
                }}
              >
                <FolderOpen size={16} />
                <span>{project.name}</span>
              </button>

              <button
                className="delete-project"
                title="删除项目"
                aria-label={`删除项目 ${project.name}`}
                onClick={() => {
                  if (window.confirm(
                    `确定删除“${project.name}”吗？`
                  )) {
                    deleteProject(project.id)
                  }
                }}
              >
                <Trash2 size={15} />
              </button>
            </div>
          ))}
        </div>

        <div className="sidebar-footer">
          V0.2 · Local Workspace
        </div>
      </aside>

      <main className="workspace">
        <header className="topbar">
          <div className="breadcrumb">
            工作台 <span>/</span> {activeProject.name}
          </div>

          {view === 'workspace' && (
            <button
              className="icon-button"
              title="切换成果面板"
              aria-label="切换成果面板"
              aria-expanded={showResults}
              onClick={() => setShowResults(value => !value)}
            >
              {showResults
                ? <PanelRightClose size={19} />
                : <PanelRightOpen size={19} />}
            </button>
          )}
        </header>

        <nav className="workspace-nav" aria-label="项目工作区">
          <button type="button" aria-pressed={view === 'workspace'} onClick={() => setView('workspace')}>创意对话</button>
          <button type="button" aria-pressed={view === 'studio'} onClick={() => setView('studio')}>Concept Studio</button>
        </nav>
        {saveFailed && <p className="workspace-save-error" role="alert">当前浏览器无法保存项目，刷新可能丢失本次更改。</p>}

        {view === 'studio' ? (
          <ConceptStudio
            key={activeProject.id}
            projectId={activeProject.id}
            projectName={activeProject.name}
            state={getProjectStudio(activeProject)}
            onChange={updateStudio}
            settings={modelSettings}
            onSettingsChange={editModelSettings}
            routing={routing}
            onTaskSettingsChange={editTaskSettings}
            onTaskReset={resetTaskSettings}
            imageSettings={imageSettings}
            onImageSettingsChange={editImageSettings}
            idea={transfer?.projectId === activeProject.id ? transfer.idea : undefined}
            onDismissIdea={() => setTransfer(null)}
          />
        ) : (
          <CreativeChat
            key={activeProject.id}
            messages={activeProject.messages}
            input={input}
            onInputChange={setInput}
            onAppend={(user, assistant) => appendChat(activeProject.id, user, assistant)}
            onTransfer={idea => {
              setTransfer({ projectId: activeProject.id, idea })
              setView('studio')
            }}
            settings={resolveTextModel('concept', routing, modelSettings)}
            onSettingsChange={editModelSettings}
          />
        )}
      </main>

      {view === 'workspace' && showResults && (
        <aside className="results-panel">
          <div className="results-header">
            <h2>设计成果</h2>
            <span>Workspace</span>
          </div>

          <div className="result-section">
            <div className="section-label">当前概念</div>
            {latestIdea ? (
              <div className="concept-card">
                <div className="concept-icon">
                  <Lightbulb size={18} />
                </div>
                <h3>{activeProject.name}</h3>
                <p>{latestIdea.content}</p>
                <span className="concept-status">
                  初步构思
                </span>
              </div>
            ) : (
              <div className="empty-result">
                输入设计想法后，概念摘要将在这里出现。
              </div>
            )}
          </div>

          <div className="result-section">
            <div className="section-label">设计流程</div>
            <div className="process-list">
              <div>01 · 问题与场景</div>
              <div>02 · 创意方向</div>
              <div>03 · 产品概念</div>
              <div>04 · 可行性验证与原创性检索</div>
              <div>05 · 设计报告</div>
            </div>
          </div>

          <div className="results-footer">
            项目内容保存在当前浏览器
          </div>
        </aside>
      )}
    </div>
  )
}

export default App

function findProvider(baseUrl: string): string {
  const normalized = baseUrl.trim().replace(/\/+$/, '')
  const found = providers.find(
    provider =>
      provider.id !== 'custom' &&
      provider.baseUrl === normalized,
  )
  return found?.id ?? 'custom'
}

export function ModelControls({
  settings,
  onSettingsChange,
  customModel,
  onModeChange,
  disabled,
  requestDescription,
  settingsOnly = false,
}: ModelControlsProps) {
  const dialog = useRef<HTMLDialogElement>(null)
  const [showKey, setShowKey] = useState(false)

  const available = canUseCustomModel(
    window.location.hostname,
  )
  const configured = hasModelSettings(settings)

  const providerId = findProvider(settings.baseUrl)
  const provider =
    providers.find(item => item.id === providerId) ??
    providers[providers.length - 1]

  function chooseProvider(id: string) {
    const selected = providers.find(item => item.id === id)
    if (!selected) return

    if (selected.id === 'custom') {
      onSettingsChange({
        baseUrl: '',
        model: '',
      })
      return
    }

    onSettingsChange({
      baseUrl: selected.baseUrl,
      model: selected.models[0] ?? '',
    })
  }

  function openSettings() {
    if (!disabled) dialog.current?.showModal()
  }

  return (
    <div className="model-controls">
      <div className="model-controls-row">
        {!settingsOnly && (
          <label>
            生成方式
            <select
              value={customModel ? 'api' : 'rules'}
              disabled={disabled}
              onChange={event => {
                const custom = event.target.value === 'api'
                onModeChange(custom)

                if (custom && !configured) {
                  dialog.current?.showModal()
                }
              }}
            >
              <option value="rules">
                本地规则（非 AI）
              </option>
              <option value="api" disabled={!available}>
                AI 模型生成
              </option>
            </select>
          </label>
        )}

        {(customModel || settingsOnly) && (
          <button
            type="button"
            disabled={disabled}
            onClick={openSettings}
          >
            AI 模型中心
          </button>
        )}

        {customModel && configured && (
          <span className="model-active">
            {settings.model.trim()}
          </span>
        )}
      </div>

      {customModel && <p>{requestDescription}</p>}

      {!available && (
        <p>
          请通过 localhost 打开工作台以使用自定义模型服务。
        </p>
      )}

      <dialog
        ref={dialog}
        className="model-dialog"
        aria-label="AI 模型中心"
      >
        <h2>AI 模型中心</h2>

        <p>
          统一配置创意生成、方案深化和语言转译所用的
          文本模型。图像模型将在下一阶段接入。
        </p>

        <div className="model-settings">
          <label>
            模型服务商
            <select
              value={providerId}
              disabled={disabled}
              onChange={event =>
                chooseProvider(event.target.value)
              }
            >
              {providers.map(item => (
                <option key={item.id} value={item.id}>
                  {item.name}
                </option>
              ))}
            </select>
          </label>

          <label>
            API 地址
            <input
              inputMode="url"
              maxLength={1000}
              disabled={disabled}
              value={settings.baseUrl}
              onChange={event =>
                onSettingsChange({
                  baseUrl: event.target.value,
                })
              }
              placeholder="https://api.example.com/v1"
            />
          </label>

          <label>
            模型 ID
            {provider.models.length > 0 && (
              <select
                value={
                  provider.models.includes(settings.model)
                    ? settings.model
                    : ''
                }
                disabled={disabled}
                onChange={event => {
                  if (event.target.value) {
                    onSettingsChange({
                      model: event.target.value,
                    })
                  }
                }}
              >
                <option value="">
                  自定义模型 ID
                </option>
                {provider.models.map(model => (
                  <option key={model} value={model}>
                    {model}
                  </option>
                ))}
              </select>
            )}

            <input
              maxLength={200}
              disabled={disabled}
              value={settings.model}
              onChange={event =>
                onSettingsChange({
                  model: event.target.value,
                })
              }
              placeholder="输入服务商支持的模型 ID"
            />
          </label>

          <label>
            API 密钥
            <input
              type={showKey ? 'text' : 'password'}
              autoComplete="off"
              maxLength={2048}
              disabled={disabled}
              value={settings.apiKey}
              onChange={event =>
                onSettingsChange({
                  apiKey: event.target.value,
                })
              }
              placeholder="输入自己的 API Key"
            />
          </label>

          <label>

              type="checkbox"
              checked={showKey}

                setShowKey(event.target.checked)

            />
            显示密钥

          <p>

            地址和模型 ID。预设模型仅为示例，

          <div className="model-controls-row">
            <span
              className="model-active"
              role="status"
            >
              {configured
                ? `已填写配置：${settings.model}`
                : '尚未完成模型配置'}
            </span>

            <button
              type="button"
              className="model-done"
              onClick={() => dialog.current?.close()}
            >
              完成
            </button>
          </div>
        </div>
      </dialog>
    </div>
  )
}

function generateGuidance(input: string): string {
  return `已记录你的设计想法：${input}

我们可以从三个方向展开：

01 · 用户与场景
明确目标用户、使用环境和核心困难。

02 · 产品与交互
寻找一种简单、直观、可验证的产品解决方式。

03 · 实现与价值
考虑材料、结构、制造可行性与设计价值。

下一步，请描述：这个产品最希望帮助哪一类人？`
}

export function CreativeChat({ messages, input, onInputChange, onAppend, onTransfer, settings, onSettingsChange }: ChatProps) {
  const [customModel, setCustomModel] = useState(false)
  const [pending, setPending] = useState(false)
  const [notice, setNotice] = useState('')
  const request = useRef<AbortController | null>(null)
  const conversation = useRef<HTMLDivElement>(null)
  const canSend = input.trim() && !pending && (!customModel || hasModelSettings(settings))

  useEffect(() => () => request.current?.abort(), [])

  useEffect(() => {
    const panel = conversation.current
    if (panel) panel.scrollTop = panel.scrollHeight
  }, [messages.length, pending])

  async function sendMessage() {
    if (!canSend || request.current) return
    const text = input.trim()
    const user: Message = { id: crypto.randomUUID(), role: 'user', content: text }
    setNotice('')

    function saveReply(reply: Pick<Message, 'content' | 'source'>) {
      onAppend(user, { id: crypto.randomUUID(), role: 'assistant', ...reply })
      onInputChange('')
    }

    if (!customModel) {
      saveReply({ content: generateGuidance(text) })
      return
    }
    if (!canUseCustomModel(window.location.hostname)) return

    const controller = new AbortController()
    request.current = controller
    setPending(true)
    try {
      const signal = AbortSignal.any([controller.signal, AbortSignal.timeout(150000)])
      const reply = await replyWithModel(messages, text, settings, signal)
      if (!controller.signal.aborted) saveReply(reply)
    } catch (error) {
      if (!controller.signal.aborted) {
        const message = error instanceof DOMException && error.name === 'TimeoutError'
          ? '等待模型回复超时。'
          : error instanceof Error && !(error instanceof TypeError)
            ? error.message
            : '无法连接模型服务。'
        setNotice(`${message} 消息仍在输入框，可重试。`)
      }
    } finally {
      if (request.current === controller) {
        request.current = null
        if (!controller.signal.aborted) setPending(false)
      }
    }
  }

  function cancelReply() {
    request.current?.abort()
    request.current = null
    setPending(false)
    setNotice('已取消回复。消息仍在输入框。')
  }

  return (
    <div className="chat-workspace">
      <div className="conversation" ref={conversation}>
        {messages.length === 0 && !pending ? (
          <div className="welcome">
            <div className="welcome-icon"><Sparkles size={25} /></div>
            <h1>今天想创造什么？</h1>
            <p>从一个问题、一段灵感，或一个尚未成形的想法开始。</p>
            <div className="suggestions">
              {[
                '设计一款帮助学生短暂放松的桌面产品',
                '探索更环保的一次性水杯',
                '为儿童设计自然声音启蒙玩具',
              ].map(suggestion => (
                <button key={suggestion} type="button" onClick={() => onInputChange(suggestion)}>
                  {suggestion}
                </button>
              ))}
            </div>
          </div>
        ) : (
          <div className="messages">
            {messages.map((message, index) => (
              <div className={`message ${message.role}`} key={message.id}>
                {message.role === 'assistant' && (
                  <div className="assistant-avatar"><Lightbulb size={16} /></div>
                )}
                <div className="message-content">
                  {message.role === 'assistant' && <div className="message-source">{modelSourceLabel(message.source)}</div>}
                  {message.content}
                  {message.role === 'assistant' && (
                    <button
                      type="button"
                      className="message-transfer"
                      disabled={pending}
                      onClick={() => onTransfer({
                        messageId: message.id,
                        question: messages[index - 1]?.role === 'user' ? messages[index - 1].content : '',
                        reply: message.content,
                        source: message.source,
                      })}
                    >
                      转为概念
                    </button>
                  )}
                </div>
              </div>
            ))}
            {pending && <div className="message user"><div className="message-content">{input.trim()}</div></div>}
          </div>
        )}
      </div>

      <div className="composer-area">
        <div className="composer">
          <div className="composer-model">
            <ModelControls
              settings={settings}
              onSettingsChange={change => {
                onSettingsChange(change)
                setNotice('')
              }}
              customModel={customModel}
              onModeChange={custom => {
                setCustomModel(custom)
                setNotice('')
              }}
              disabled={pending}
              requestDescription="发送时会将本轮消息和当前项目的近期对话发送给所选服务。"
            />
          </div>
          <textarea
            value={input}
            onChange={event => onInputChange(event.target.value)}
            placeholder="描述你的设计问题或创意想法..."
            rows={3}
            readOnly={pending}
            onKeyDown={event => {
              if (event.key === 'Enter' && !event.shiftKey && !event.nativeEvent.isComposing) {
                event.preventDefault()
                void sendMessage()
              }
            }}
          />
          <div className="composer-bottom">
            <span>Shift + Enter 换行</span>
            {pending ? (
              <button type="button" onClick={cancelReply} aria-label="取消回复"><Square size={15} /></button>
            ) : (
              <button type="button" onClick={sendMessage} disabled={!canSend} aria-label="发送想法"><ArrowUp size={19} /></button>
            )}
          </div>
        </div>
        <p className="composer-note" role="status">
          {pending ? `正在等待 ${settings.model.trim()} 回复…`
            : notice || (customModel && !hasModelSettings(settings) ? '请完成 API 设置后发送。' : '')}
        </p>
      </div>
    </div>
  )
}

          {Object.entries(validationOutcomes).map(([key, label]) => <option key={key} value={key}>{label}</option>)}
        </select>
      </label>
      <p id="validation-help" className="studio-hint">
        {ready ? '对照通过标准和实测记录，由你填写验证结果。' : '填写通过标准与实测记录后，可给出验证结果。'}
      </p>
      <label className="studio-field">人工可行性分析结论
        <textarea rows={4} maxLength={4000} value={developmentOf(concept).feasibilityConclusion} placeholder="修正 AI 分析，区分可行条件、剩余风险与未实测事项。"
          onChange={event => {
            const feasibilityConclusion = event.target.value
            onChange(current => ({ ...current, development: { ...developmentOf(current), feasibilityConclusion } }))
          }} />
      </label>
      <p className="studio-hint">AI 只提出分析和测试方案，不会填写实测记录或替你判定达到标准。概念冻结不代表实测通过。</p>
    </section>
  )
}

function ReportPanel({ projectName, round, roundNumber, concept }: Pick<RefinementProps, 'projectName' | 'round' | 'roundNumber' | 'concept'>) {
  const report = buildDesignReport(projectName, round, roundNumber, concept)

  function download() {
    const file = new Blob([formatDesignReport(report)], { type: 'text/markdown;charset=utf-8' })
    const url = URL.createObjectURL(file)
    const link = document.createElement('a')
    link.href = url
    link.download = `INNO-LIT-第${roundNumber}轮-设计报告.md`
    document.body.append(link)
    link.click()
    link.remove()
    setTimeout(() => URL.revokeObjectURL(url), 1000)
  }

  return (
    <section className="studio-surface studio-refinement" aria-labelledby="report-heading">
      <header className="studio-report-header">
        <div>
          <h2 id="report-heading">设计报告</h2>
          <p>报告使用当前轮次和选定方案的最新记录。</p>
        </div>
        <button type="button" className="studio-button primary" onClick={download}>
          <Download size={16} aria-hidden="true" /> 下载 Markdown
        </button>
      </header>
      {report.sections.map(section => (
        <section className="studio-report-section" key={section.heading}>
          <h3>{section.heading}</h3>
          <dl>
            {section.entries.map(entry => (
              <div key={entry.url ?? entry.label}>
                <dt>{entry.label}</dt>
                <dd>{entry.value.trim() || '未填写'}</dd>
                {entry.url && <dd><a href={entry.url} target="_blank" rel="noreferrer">查看来源</a></dd>}
              </div>
            ))}
          </dl>
        </section>
      ))}
    </section>
  )
}

export function RefinementWorkspace({ projectId, projectName, round, roundNumber, concept, onChange, children, imageSettings, settings, routing, onTaskSettingsChange, onTaskReset, defaultTextSettings, defaultImageSettings, translationSettings }: RefinementProps) {
  const [stage, setStage] = useState<Stage>('notes')
  const locked = Boolean(developmentOf(concept).frozen)
  function edit(update: (current: Concept) => Concept) {
    onChange(current => developmentOf(current).frozen ? current : update(current))
  }

  return (
    <>
      <details className="studio-surface development-model-settings">
        <summary>AI 模型中心</summary>
        <TaskModelCenter routing={routing} defaultText={defaultTextSettings} defaultImage={defaultImageSettings} onChange={onTaskSettingsChange} onReset={onTaskReset} />
      </details>
      {locked && <p className="studio-badge">冻结 V{developmentOf(concept).frozen!.revision} · 设计内容已锁定，展示制作引用此版本</p>}
      <nav className="studio-refinement-nav" aria-label="深化工作区">
        {stageNames.map(key => (
          <button
            type="button"
            key={key}
            className="studio-button"
            aria-pressed={stage === key}
            onClick={() => setStage(key)}
          >
            {stages[key]}
          </button>
        ))}
      </nav>
      {stage === 'notes' && (
        <DevelopmentAssistant step="notes" concept={concept} settings={settings} onChange={edit}>
        <section className="studio-surface studio-refinement" aria-labelledby="notes-heading">
          <div>
            <h2 id="notes-heading">首轮验证建议</h2>
            <p><strong>关键风险</strong>　{concept.sentences.risk}</p>
            <p><strong>验证方式</strong>　{concept.sentences.test}</p>
          </div>
          <label className="studio-field">
            深化笔记
            <textarea
              rows={5}
              maxLength={4000}
              value={concept.notes}
              onChange={event => {
                const notes = event.target.value
                edit(current => ({ ...current, notes }))
              }}
              placeholder="记录方案调整、待解决的问题与下一步"
            />
          </label>
          {children}
        </section>
        </DevelopmentAssistant>
      )}
      {stage === 'validation' && <DevelopmentAssistant step="validation" concept={concept} settings={settings} onChange={edit}>
        <ValidationPanel concept={concept} onChange={edit} />
      </DevelopmentAssistant>}
      {stage === 'research' && (
        <DevelopmentAssistant step="research" concept={concept} settings={settings} onChange={edit}>
        <EvidencePanel
          research={concept.research ?? emptyResearch}
          onChange={update => edit(current => ({ ...current, research: update(current.research ?? emptyResearch) }))}
        />
        <section className="studio-surface studio-refinement"><label className="studio-field">人工检索比较结论
          <textarea rows={4} maxLength={4000} value={developmentOf(concept).researchConclusion} placeholder="阅读来源后修正相似与差异、检索范围和仍需调查的部分。"
            onChange={event => { const researchConclusion = event.target.value; edit(current => ({ ...current, development: { ...developmentOf(current), researchConclusion } })) }} />
        </label></section>
        </DevelopmentAssistant>
      )}
      {stage === 'optimize' && <ConceptFreezePanel concept={concept} settings={settings} onChange={onChange} onFrozen={() => setStage('visuals')} />}
      {stage === 'report' && <ReportPanel projectName={projectName} round={round} roundNumber={roundNumber} concept={concept} />}
      {stage === 'visuals' && <VisualStudio key={`${roundNumber}:${concept.id}`} projectId={projectId}
        projectName={projectName} round={round} roundNumber={roundNumber} concept={concept} onChange={onChange}
        imageSettings={imageSettings}
        modelSettings={translationSettings} 
         />}
    </>
  )
}
